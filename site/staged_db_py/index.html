    LIST_JOBS_NEW module-attribute 

    LIST_JOBS_NEW = '\nSELECT sj.job_id as job_id, \n       extract(epoch from sj.created) as created,\n       (SELECT count(*) FROM files WHERE files.job_id=sj.job_id AND ready) as staged_files, \n       sj.total_files as total_files, \n       sj.completed as completed, \n       (SELECT extract(epoch from max(f.readytime)) FROM files as f WHERE f.job_id=sj.job_id) as last_readytime,\n       (SELECT count(*) FROM files WHERE files.job_id=sj.job_id AND error) as error_files,\n       (SELECT filename FROM files WHERE files.job_id=sj.job_id LIMIT 1) as filename\nFROM staging_jobs as sj\nORDER BY sj.job_id;\n'

    def initdb(config):
    """Initialise a connection pool to the database.

    Args:
        config: A Config object containing database connection paramters

    Returns:
        None
    """
    global DBPOOL
    DBPOOL = psycopg_pool.ConnectionPool('postgresql://%s:%s@%s/%s' % (config.DBUSER, config.DBPASSWORD, config.DBHOST, config.DBNAME),
                                         min_size=MINCONN,
                                         max_size=MAXCONN,
                                         open=True,
                                         timeout=360,)
